126Geometry* geometry_create_sphere(float R, unsigned int lat_amount, unsigned int long_amount)
127{
128 if (R <= 0.f)
129 {
130 LOG_ERROR("Sphere radius cannot equal to zero.");
131 return NULL;
132 }
133
134 unsigned int vertices_amount = (lat_amount + 1) * (long_amount + 1);
135 unsigned int indices_amount = lat_amount * long_amount * 6;
136
137 size_t vertices_size = sizeof(Vertex) * vertices_amount;
138
139 Vertex* vertices = malloc(vertices_size);
140 unsigned int* indices = malloc(sizeof(unsigned int) * indices_amount);
141
142 float step_lat = M_PI / lat_amount;
143 float step_long = (2.f * M_PI) / long_amount;
144
145 unsigned int vertex_i = 0;
146
147 for (unsigned int i = 0; i <= lat_amount; ++i)
148 {
149 float lat_angle = i * step_lat;
150
151 for (unsigned int j = 0; j <= long_amount; ++j)
152 {
153 float long_angle = j * step_long;
154
155 float x = R * sinf(lat_angle) * cosf(long_angle);
156 float y = R * sinf(lat_angle) * sinf(long_angle);
157 float z = R * cosf(lat_angle);
158
159 vertices[vertex_i].pos.x = x;
160 vertices[vertex_i].pos.y = y;
161 vertices[vertex_i].pos.z = z;
162
163 ++vertex_i;
164 }
165 }
166
167 unsigned int indice_i = 0;
168
169 for (unsigned int i = 0; i < lat_amount; ++i)
170 {
171 for (unsigned int j = 0; j < long_amount; ++j)
172 {
173 unsigned int k1 = i * (long_amount + 1) + j;
174 unsigned int k2 = k1 + long_amount + 1;
175
176 indices[indice_i++] = k1;
177 indices[indice_i++] = k2;
178 indices[indice_i++] = k1 + 1;
179
180 indices[indice_i++] = k1 + 1;
181 indices[indice_i++] = k2;
182 indices[indice_i++] = k2 + 1;
183 }
184 }
185
186 VertexLayout layout;
187 vertex_layout_init_default(&layout);
188
189 return geometry_create(vertices, vertices_amount,
190 indices, indices_amount, layout);
