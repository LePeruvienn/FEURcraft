144void shader_program_link(ShaderProgram* program)
145{

147
148 if(program->shaders[SHADER_TYPE_VERT] == NULL ||

151 LOG_ERROR("Cannot link a program without a Vertex and a Fragment Shader.");
152 return;
153 }
154
155 if (program->status == PROGRAM_STATUS_LINKED)
156 {
157 LOG_WARNING("Program was already linked, relink program ...");
158 }
159
160 GL_CALL(glLinkProgram(program->id));
161
162 int success = get_program_status(program->id, GL_LINK_STATUS);
163 program->status = (success == 0) ?
