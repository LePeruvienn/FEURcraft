34AudioSource AUDIO_SOURCE_CREATE(const char *filename)
35{
36 SF_INFO sfinfo = {0};
37 SNDFILE *sndfile = sf_open(filename, SFM_READ, &sfinfo);
38
39 if (!sndfile) {
40 fprintf(stderr, "Failed to open sound file '%s': %s\n", filename, sf_strerror(NULL));
41 return 0;
42 }
43
44 /* Total raw frames in the file */
45 sf_count_t num_frames = sfinfo.frames;
46 sf_count_t total_samples = num_frames * sfinfo.channels;
47
48 short *pcm_data = (short *)malloc(total_samples * sizeof(short));
49 if (!pcm_data) {
50 fprintf(stderr, "Memory allocation error\n");
51 sf_close(sndfile);
52 return 0;
53 }
54
55 /* Read all audio data as 16-bit PCM */
56 sf_count_t read_count = sf_read_short(sndfile, pcm_data, total_samples);
57 sf_close(sndfile);
58
59 short *mono_data = NULL;
60 sf_count_t mono_samples = 0;
61
62 if (sfinfo.channels == 1) {
63 /* Already mono: use data directly */
64 mono_data = pcm_data;
65 mono_samples = read_count;
66 }
67 else if (sfinfo.channels == 2) {
68 /* Downmix stereo to mono in-memory */
69 mono_samples = num_frames;
70 mono_data = (short *)malloc(mono_samples * sizeof(short));
71
72 for (sf_count_t i = 0; i < mono_samples; i++) {
73 int left = pcm_data[i * 2];
74 int right = pcm_data[i * 2 + 1];
75
76 /* Average L & R channels (divide first or sum as int to avoid 16-bit overflow) */
77 mono_data[i] = (short)((left + right) / 2);
78 }
79
80 /* Free original stereo raw buffer */
81 free(pcm_data);
82 }
83 else {
84 fprintf(stderr, "Unsupported channel count: %d\n", sfinfo.channels);
85 free(pcm_data);
86 return 0;
87 }
88
89 /* Upload the buffer as AL_FORMAT_MONO16 */
90 ALuint buffer;
91 alGenBuffers(1, &buffer);
92 alBufferData(buffer, AL_FORMAT_MONO16, mono_data,
93 (ALsizei)(mono_samples * sizeof(short)),
94 sfinfo.samplerate);
95
96 free(mono_data);
97
98 return buffer;
99}
